include<reg52.h>

#define uint unsigned int

#define uchar unsigned char

sbit dula = P2^6;

sbit wela = P2^7;

sbit key1 = P3^4;

sbit d1 = P1^0;



uchar code table[] = {

          0x3f,0x06,0x5b,0x4f,

          0x66,0x6d,0x7d,0x07,

          0x7f,0x6f

};



uchar num =  0;



void delayms(uint ms)

{

    uint i,j;

    for(i = ms; i>0;i--)

        for(j = 110;j>0;j--)

}



viod main()

{

    wela = 1;

    P0 = 0xfe;

    wela = 0;

    P3 = 0xff;

    while(1)

    {

        if(key1 == 0)

        {

            delayms(5);

            if(key1 == 0)

            {

                d1 = 0;

                num++;

                if(num >= 10)

                num = 0;

                while (!key1);

                delayms(5);

            }

     }

    else

    {

        d1 = 1;

    }

        dula = 1;

        P0 = table[num];

        dula = 0;

    }

}
